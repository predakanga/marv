458	    private async Task HandleCap(IrcMessage message, CancellationToken ct)
459	    {
460	        if (message.Parameters.Count < 3) return;
461	
462	        var subCommand = message.Parameters[1].ToUpperInvariant();
463	        var isMultiLine = message.Parameters.Count > 3 && message.Parameters[2] == "*";
464	        var capList = message.Parameters[isMultiLine ? 3 : 2];
465	
466	        switch (subCommand)
467	        {
468	            case "LS":
469	                foreach (var cap in ParseCapList(capList))
470	                {
471	                    _capabilityManager.SetAvailable(cap.Key, cap.Value);
472	                }
473	
474	                if (!isMultiLine)
475	                {
476	                    // All capabilities listed — request the ones we want
477	                    var toRequest = DesiredCapabilities
478	                        .Where(c => _capabilityManager.IsAvailable(c))
479	                        .ToList();
480	
481	                    if (toRequest.Count > 0)
482	                    {
483	                        _logger.LogInformation("Requesting capabilities: {Caps}",
484	                            string.Join(", ", toRequest));
485	                        await SendRawAsync(new IrcMessage("CAP", ["REQ", string.Join(' ', toRequest)]), ct);
486	                    }
487	                    else
488	                    {
489	                        await SendRawAsync(new IrcMessage("CAP", ["END"]), ct);
490	        
491	                    }
492	                }
493	                break;
494	
495	            case "ACK":
496	                foreach (var capName in capList.Split(' ', StringSplitOptions.RemoveEmptyEntries))
497	                {
498	                    _capabilityManager.SetNegotiated(capName);
499	                    _logger.LogDebug("Capability acknowledged: {Cap}", capName);
500	                }
501	
502	                // Start SASL if negotiated and credentials are configured
503	                if (_capabilityManager.IsNegotiated(Platform.Capabilities.Sasl)
504	                    && !string.IsNullOrEmpty(_config.SaslUser) && !string.IsNullOrEmpty(_config.SaslPassword)
505	                    && !_saslInProgress)
506	                {
507	                    _saslInProgress = true;
508	                    await SendRawAsync(new IrcMessage("AUTHENTICATE", ["PLAIN"]), ct);
509	                }
510	                else
511	                {
512	                    await SendRawAsync(new IrcMessage("CAP", ["END"]), ct);
513	    
514	                }
515	                break;
516	
517	            case "NAK":
518	                _logger.LogWarning("Capabilities rejected: {Caps}", capList);
519	                await SendRawAsync(new IrcMessage("CAP", ["END"]), ct);
520	
521	                break;
522	
523	            case "NEW":
524	                foreach (var cap in ParseCapList(capList))
525	                {
526	                    _capabilityManager.AddNewCapability(cap.Key, cap.Value);
527	                }
528	                var newCaps = ParseCapList(capList)
529	                    .Where(c => DesiredCapabilities.Contains(c.Key, StringComparer.OrdinalIgnoreCase))
530	                    .Select(c => c.Key)
531	                    .ToList();
532	                if (newCaps.Count > 0)
533	                {
534	                    await SendRawAsync(new IrcMessage("CAP", ["REQ", string.Join(' ', newCaps)]), ct);
535	                }
536	
537	                var capChangedNew = CreateCapChangedEvent(message);
538	                await FanOutEventAsync(capChangedNew, ct);
539	                break;
540	
541	            case "DEL":
542	                foreach (var capName in capList.Split(' ', StringSplitOptions.RemoveEmptyEntries))
543	                {
544	                    _capabilityManager.RemoveCapability(capName);
545	                    _logger.LogDebug("Capability removed: {Cap}", capName);
546	                }
547	
548	                var capChangedDel = CreateCapChangedEvent(message);
549	                await FanOutEventAsync(capChangedDel, ct);
550	                break;
551	        }
552	    }

676	    /// Timeout (in seconds) for post-registration authentication to complete.
677	    /// </summary>
678	    private const int AuthTimeoutSeconds = 15;
679	
680	    /// <summary>
681	    /// Waits for all pending auth steps to complete, then fires ReadyEvent and
682	    /// joins channels. If auth doesn't complete within the timeout, proceeds anyway.
683	    /// </summary>
684	    private async Task CompleteAuthSequenceAsync(CancellationToken ct)
685	    {
686	        using var timeoutCts = CancellationTokenSource.CreateLinkedTokenSource(ct);
687	        timeoutCts.CancelAfter(TimeSpan.FromSeconds(AuthTimeoutSeconds));
688	
689	        try
690	        {
691	            await _readyTcs!.Task.WaitAsync(timeoutCts.Token);
692	        }
693	        catch (OperationCanceledException) when (!ct.IsCancellationRequested)
694	        {
695	            _logger.LogWarning("Post-registration authentication timed out after {Seconds}s — proceeding", AuthTimeoutSeconds);
696	        }
697	
698	        var syntheticMessage = new IrcMessage(null, null, "READY", []);
699	        await FireReadyAndJoinAsync(syntheticMessage, ct);
700	    }

775	    {
776	        var spaceIdx = ctcpContent.IndexOf(' ');
777	        var command = spaceIdx >= 0 ? ctcpContent[..spaceIdx].ToUpperInvariant() : ctcpContent.ToUpperInvariant();
778	        var args = spaceIdx >= 0 ? ctcpContent[(spaceIdx + 1)..] : null;
779	
780	        switch (command)
781	        {
782	            case "ACTION":
783	            {
784	                var isChannel = IsChannelName(target);
785	                var channel = isChannel ? GetChannel(target) : null;
786	                var evt = new ActionEvent
787	                {
788	                    Timestamp = GetTimestamp(message),
789	                    RawMessage = message,
790	                    MessageId = message.Tags.GetValueOrDefault("msgid"),
791	                    BatchId = message.Tags.GetValueOrDefault("batch"),
792	                    Channel = channel,
793	                    Sender = sender,
794	                    Text = args ?? ""
795	                };
796	                await FanOutEventAsync(evt, ct);
797	                break;
798	            }
799	            case "VERSION":
800	                await SendRawAsync(new IrcMessage("NOTICE", [sender.Nick,
801	                    "\x01VERSION Marv IRC Bot 0.1.0\x01"]), ct);
802	                break;
803	            case "PING":
804	                await SendRawAsync(new IrcMessage("NOTICE", [sender.Nick,
805	                    $"\x01PING {args ?? ""}\x01"]), ct);
806	                break;
807	            case "TIME":
808	                await SendRawAsync(new IrcMessage("NOTICE", [sender.Nick,
809	                    $"\x01TIME {DateTimeOffset.UtcNow:R}\x01"]), ct);
810	                break;
811	            default:
812	            {
813	                var evt = new CtcpEvent
814	                {
815	                    Timestamp = GetTimestamp(message),
816	                    RawMessage = message,
817	                    MessageId = message.Tags.GetValueOrDefault("msgid"),
818	                    BatchId = message.Tags.GetValueOrDefault("batch"),
819	                    Sender = sender,
820	                    Command = command,
821	                    Args = args,
822	                    IsDirect = !IsChannelName(target)
823	                };
824	                await FanOutEventAsync(evt, ct);
825	                break;
826	            }
827	        }
828	    }

889	    private async Task HandleJoin(IrcMessage message, CancellationToken ct)
890	    {
891	        if (message.Source?.Nick is null || message.Parameters.Count < 1) return;
892	
893	        var channelName = message.Parameters[0];
894	        var user = GetOrCreateUser(message.Source);
895	
896	        // extended-join: JOIN #channel account :realname
897	        string? account = null;
898	        if (_capabilityManager.IsNegotiated(Platform.Capabilities.ExtendedJoin) && message.Parameters.Count >= 2)
899	        {
900	            account = message.Parameters[1] == "*" ? null : message.Parameters[1];
901	            if (account is not null) user.Account = account;
902	            if (message.Parameters.Count >= 3)
903	                user.RealName = message.Parameters[2];
904	        }
905	
906	        // Account from account-tag
907	        if (message.Tags.TryGetValue("account", out var acctTag) && acctTag is not null)
908	        {
909	            account = acctTag == "*" ? null : acctTag;
910	            if (account is not null) user.Account = account;
911	        }
912	
913	        var comparer = CaseMapping.GetComparer(_serverInfo.CaseMapping);
914	
915	        if (comparer.Equals(user.Nick, _currentNick))
916	        {
917	            // Bot is joining a channel
918	            var channel = new IrcChannel(channelName, comparer);
919	            channel.AddMember(user);
920	            _channels[channelName] = channel;
921	            _self = user;
922	
923	            var evt = new UserJoinedEvent
924	            {
925	                Timestamp = GetTimestamp(message),
926	                RawMessage = message,
927	                MessageId = message.Tags.GetValueOrDefault("msgid"),
928	                BatchId = message.Tags.GetValueOrDefault("batch"),
929	                Channel = channel,
930	                User = user,
931	                Account = account
932	            };
933	            await FanOutEventAsync(evt, ct);
934	        }
935	        else
936	        {
937	            var channel = GetChannel(channelName);
938	            if (channel is not null)
939	            {
940	                channel.AddMember(user);
941	                user.AddChannel(channel);
942

1512	    private List<ModeChange> ParseModeChanges(string modeString, IReadOnlyList<string> parameters, int paramStart, IrcChannel channel)
1513	    {
1514	        var changes = new List<ModeChange>();
1515	        var adding = true;
1516	        var paramIdx = paramStart;
1517	
1518	        foreach (var c in modeString)
1519	        {
1520	            if (c == '+') { adding = true; continue; }
1521	            if (c == '-') { adding = false; continue; }
1522	
1523	            string? param = null;
1524	            var modeType = GetModeType(c);
1525	
1526	            // Type A: always has parameter
1527	            // Type B: always has parameter
1528	            // Type C: has parameter when setting, not when unsetting
1529	            // Type D: never has parameter
1530	            // Prefix modes: always have parameter (the nick)
1531	            if (_serverInfo.Prefix.HasMode(c))
1532	            {
1533	                if (paramIdx < parameters.Count)
1534	                    param = parameters[paramIdx++];
1535	            }
1536	            else if (modeType is 'A' or 'B')
1537	            {
1538	                if (paramIdx < parameters.Count)
1539	                    param = parameters[paramIdx++];
1540	            }
1541	            else if (modeType == 'C' && adding)
1542	            {
1543	                if (paramIdx < parameters.Count)
1544	                    param = parameters[paramIdx++];
1545	            }
1546	
1547	            changes.Add(new ModeChange { IsSet = adding, Mode = c, Parameter = param });
1548	        }
1549	
1550	        return changes;
1551	    }

1554	    {
1555	        var chanModes = _serverInfo.ChannelModes;

1557	        if (chanModes.TypeB.Contains(mode)) return 'B';
1558	        if (chanModes.TypeC.Contains(mode)) return 'C';
1559	        if (chanModes.TypeD.Contains(mode)) return 'D';

/// <summary>
/// Token bucket rate limiter for outbound IRC messages.
/// Allows a configurable burst of messages, then throttles to a steady refill rate.
/// Can be disabled entirely for testing or unrestricted environments.
/// </summary>
internal sealed class TokenBucketRateLimiter
{
    private readonly bool _enabled;
    private readonly int _burstLimit;
    private readonly double _refillRatePerSecond;
    private double _tokens;
    private DateTimeOffset _lastRefill;
    private readonly object _lock = new();

    /// <summary>
    /// Creates a new token bucket rate limiter.
    /// </summary>
    /// <param name="enabled">Whether rate limiting is active. When false, <see cref="WaitAsync"/> returns immediately.</param>
    /// <param name="burstLimit">Maximum tokens (messages) available at once.</param>
    /// <param name="refillRatePerSecond">Tokens added per second (e.g. 0.5 = one token every 2 seconds).</param>
    public TokenBucketRateLimiter(bool enabled = true, int burstLimit = 5, double refillRatePerSecond = 0.5)
    {
        _enabled = enabled;
        _burstLimit = burstLimit;
        _refillRatePerSecond = refillRatePerSecond;
        _tokens = burstLimit;
        _lastRefill = DateTimeOffset.UtcNow;
    }

    /// <summary>Whether rate limiting is active.</summary>
    public bool Enabled => _enabled;

    /// <summary>Maximum burst size.</summary>
    public int BurstLimit => _burstLimit;

    /// <summary>Token refill rate per second.</summary>
    public double RefillRatePerSecond => _refillRatePerSecond;

    /// <summary>
    /// Waits until a token is available, consuming it before returning.
    /// Returns immediately when rate limiting is disabled.
    /// </summary>
    public async Task WaitAsync(CancellationToken ct)
    {
        if (!_enabled) return;

        while (true)
        {
            ct.ThrowIfCancellationRequested();

            lock (_lock)
            {
                var now = DateTimeOffset.UtcNow;
                var elapsed = (now - _lastRefill).TotalSeconds;
                _tokens = Math.Min(_burstLimit, _tokens + elapsed * _refillRatePerSecond);
                _lastRefill = now;

                if (_tokens >= 1.0)
                {
                    _tokens -= 1.0;
                    return;
                }
            }

            await Task.Delay(100, ct);
        }
    }

    /// <summary>
    /// Resets the token bucket to full capacity. Called when a new connection is established.
    /// </summary>
    public void Reset()
    {
        lock (_lock)
        {
            _tokens = _burstLimit;
            _lastRefill = DateTimeOffset.UtcNow;
        }
    }
}
