#!/usr/bin/env bash
# Prepares a release: bumps version, updates CHANGELOG.md, commits, and tags.
#
# Usage: ./scripts/prepare-release.sh [major|minor|patch]
#
# If no argument is given, prompts interactively for the bump type.
# The [Unreleased] section in CHANGELOG.md must have content to release.

set -euo pipefail

REPO_ROOT=$(git rev-parse --show-toplevel)
cd "$REPO_ROOT"

# --- Validate preconditions ---

if [[ -n $(git status --porcelain) ]]; then
    echo "Error: working tree is not clean. Commit or stash changes first." >&2
    exit 1
fi

UNRELEASED_CONTENT=$(sed -n '/^## \[Unreleased\]/,/^## \[/{ /^## \[/d; /^$/d; p; }' CHANGELOG.md)
if [[ -z "$UNRELEASED_CONTENT" ]]; then
    echo "Error: no content under [Unreleased] in CHANGELOG.md." >&2
    exit 1
fi

# --- Read current version ---

CURRENT=$(grep -oP '(?<=<Version>)[^<]+' Directory.Build.props)
IFS='.' read -r MAJOR MINOR PATCH <<< "$CURRENT"

NEXT_MAJOR="$((MAJOR + 1)).0.0"
NEXT_MINOR="${MAJOR}.$((MINOR + 1)).0"
NEXT_PATCH="${MAJOR}.${MINOR}.$((PATCH + 1))"

# --- Determine bump type ---

bump_type="${1:-}"

if [[ -z "$bump_type" ]]; then
    echo "Current version: ${CURRENT}"
    echo ""
    echo "  1) patch  → ${NEXT_PATCH}"
    echo "  2) minor  → ${NEXT_MINOR}"
    echo "  3) major  → ${NEXT_MAJOR}"
    echo ""
    read -rp "Bump type [1/2/3]: " choice
    case "$choice" in
        1|patch) bump_type=patch ;;
        2|minor) bump_type=minor ;;
        3|major) bump_type=major ;;
        *) echo "Invalid choice." >&2; exit 1 ;;
    esac
fi

case "$bump_type" in
    patch) VERSION="$NEXT_PATCH" ;;
    minor) VERSION="$NEXT_MINOR" ;;
    major) VERSION="$NEXT_MAJOR" ;;
    *) echo "Error: argument must be 'major', 'minor', or 'patch'." >&2; exit 1 ;;
esac

TAG="v${VERSION}"
DATE=$(date +%Y-%m-%d)

if git rev-parse "$TAG" >/dev/null 2>&1; then
    echo "Error: tag ${TAG} already exists." >&2
    exit 1
fi

echo "Releasing ${TAG}..."

# --- Update files ---

sed -i "s|<Version>[^<]*</Version>|<Version>${VERSION}</Version>|" Directory.Build.props

sed -i "s/^## \[Unreleased\]$/## [Unreleased]\n\n## [${VERSION}] - ${DATE}/" CHANGELOG.md

PREV_TAG=$(grep -oP '^\[[\d.]+\]' CHANGELOG.md | head -2 | tail -1 | tr -d '[]')
sed -i "/^\[${PREV_TAG}\]:/ i\\[${VERSION}]: https://github.com/predakanga/marv/compare/v${PREV_TAG}...v${VERSION}" CHANGELOG.md

# --- Commit and tag ---

git add Directory.Build.props CHANGELOG.md
git commit -m "Prepare ${TAG} release"
git tag -a "$TAG" -m "$TAG"

echo ""
echo "Release ${TAG} prepared. Push with:"
echo "  git push --follow-tags"

The Dockerfile already does the csproj-copy-then-restore pattern for layer caching, but that only helps with repeated local builds. For CI with Buildx and GHA cache, the layer cache (cache-from: type=gha) should handle it — but the restore layer gets invalidated whenever any csproj changes.

The real win here is using BuildKit cache mounts to persist the NuGet package cache across builds, even when the restore layer is invalidated.

1	# Release Runbook
2	
3	This document describes how to create a new release of Marv.
4	
5	## Prerequisites
6	
7	- Push access to the repository
8	- All CI checks passing on `main`
9	
10	## Steps
11	
12	### 1. Decide on a version number
13	
14	Marv uses [Semantic Versioning](https://semver.org/):
15	
16	- **Major** (`X.0.0`): Breaking changes to the plugin API or configuration format
17	- **Minor** (`0.X.0`): New features, new plugins, new configuration options
18	- **Patch** (`0.0.X`): Bug fixes, dependency updates, documentation changes
19	
20	While Marv is pre-1.0, minor version bumps may include breaking changes.
21	
22	### 2. Verify main is ready
23	
24	```bash
25	git checkout main
26	git pull
27	dotnet build -c Release
28	dotnet test -c Release --filter "Category!=Integration"
29	```
30	
31	Ensure all tests pass and the build is clean.
32	
33	### 3. Bump the version number
34	
35	Update the version in these locations:
36	
37	- **`Directory.Build.props`** — the `<Version>` property (sets assembly and
38	  package versions for all projects)
39	- **`Dockerfile`** — the `ARG VERSION=` default value
40	- **`CHANGELOG.md`** — rename the `[Unreleased]` section to
41	  `[X.Y.Z] - YYYY-MM-DD` and add a comparison link at the bottom of the file
42	
43	Commit these changes to `main` before tagging.
44	
45	### 4. Create and push a version tag
46	
47	```bash
48	git tag vX.Y.Z
49	git push origin vX.Y.Z
50	```
51	
52	Replace `vX.Y.Z` with the version number you chose.
53	
54	### 5. Monitor the release workflow
55	
56	The `Release` workflow triggers automatically on version tags. It will:
57	
58	1. Run the full test suite
59	2. Build platform-specific binaries (linux-x64, linux-arm64, osx-x64, osx-arm64, win-x64)
60	3. Build and push a multi-arch Docker image to `ghcr.io`
61	4. Create a GitHub Release with auto-generated release notes and attached binaries
62	
63	Monitor the workflow at **Actions > Release** in the GitHub UI.
64	
65	### 6. Review the GitHub Release
66	
67	Once the workflow completes:
68	
69	1. Go to the **Releases** page
70	2. Review the auto-generated release notes — edit if needed to highlight important changes
71	3. If this is a pre-release, check the **Set as a pre-release** box
72	
73	### 7. Verify the Docker image
74	
75	```bash
76	docker pull ghcr.io/predakanga/marv:X.Y.Z
77	docker run --rm ghcr.io/predakanga/marv:X.Y.Z --help
78	```
79	
80	
81	Replace `X.Y.Z` with the release version.
82	
83	## Rolling back a release
84	
85	If a release has a critical issue:
86	
87	1. Delete the GitHub Release from the Releases page
88	2. Delete the tag:
89	   ```bash
90	   git tag -d vX.Y.Z
91	   git push origin :refs/tags/vX.Y.Z
92	   ```
93	3. Fix the issue, then re-release following the steps above
94	
95	The Docker image tag will be overwritten by the new release. The `:latest` tag always points to the most recent release.
96	
97	## Hotfix releases
98	
99	For urgent fixes against an older release:
100	
101	1. Create a branch from the release tag:
102	   ```bash
103	   git checkout -b hotfix/vX.Y.Z vX.Y.Z
104	   ```
105	2. Apply the fix and push the branch
106	3. Tag and push:
107	   ```bash
108	   git tag vX.Y.Z
109	   git push origin vX.Y.Z
110	   ```
111	4. Cherry-pick the fix back to `main` if applicable
