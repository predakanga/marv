225	    /// Computes the Levenshtein edit distance between two strings.
226	    /// </summary>
227	    private static int LevenshteinDistance(string a, string b)
228	    {
229	        if (a.Length == 0) return b.Length;
230	        if (b.Length == 0) return a.Length;
231	
232	        var prev = new int[b.Length + 1];
233	        var curr = new int[b.Length + 1];
234	
235	        for (var j = 0; j <= b.Length; j++)
236	            prev[j] = j;
237	
238	        for (var i = 1; i <= a.Length; i++)
239	        {
240	            curr[0] = i;
241	            for (var j = 1; j <= b.Length; j++)
242	            {
243	                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
244	                curr[j] = Math.Min(
245	                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
246	                    prev[j - 1] + cost);
247	            }
248	
249	            (prev, curr) = (curr, prev);
250	        }
251	
252	        return prev[b.Length];
253	    }

    /// </summary>
    private static int LevenshteinDistance(string a, string b)
    {
        if (a.Length == 0) return b.Length;
        if (b.Length == 0) return a.Length;

        var prev = new int[b.Length + 1];
        var curr = new int[b.Length + 1];

        for (var j = 0; j <= b.Length; j++)
            prev[j] = j;

        for (var i = 1; i <= a.Length; i++)
        {
            curr[0] = i;
            for (var j = 1; j <= b.Length; j++)
            {
                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
                curr[j] = Math.Min(
                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
                    prev[j - 1] + cost);
            }

            (prev, curr) = (curr, prev);
        }

        var prev = new int[b.Length + 1];
        var curr = new int[b.Length + 1];

        for (var j = 0; j <= b.Length; j++)
            prev[j] = j;

        for (var i = 1; i <= a.Length; i++)
        {
            curr[0] = i;
            for (var j = 1; j <= b.Length; j++)
            {
                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
                curr[j] = Math.Min(
                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
                    prev[j - 1] + cost);
            }

            (prev, curr) = (curr, prev);
        }

        return prev[b.Length];
    }
