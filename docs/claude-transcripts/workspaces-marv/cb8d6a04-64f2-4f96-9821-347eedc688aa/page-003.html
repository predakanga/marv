225	    /// Computes the Levenshtein edit distance between two strings.
226	    /// </summary>
227	    private static int LevenshteinDistance(string a, string b)
228	    {
229	        if (a.Length == 0) return b.Length;
230	        if (b.Length == 0) return a.Length;
231	
232	        var prev = new int[b.Length + 1];
233	        var curr = new int[b.Length + 1];
234	
235	        for (var j = 0; j <= b.Length; j++)
236	            prev[j] = j;
237	
238	        for (var i = 1; i <= a.Length; i++)
239	        {
240	            curr[0] = i;
241	            for (var j = 1; j <= b.Length; j++)
242	            {
243	                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
244	                curr[j] = Math.Min(
245	                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
246	                    prev[j - 1] + cost);
247	            }
248	
249	            (prev, curr) = (curr, prev);
250	        }
251	
252	        return prev[b.Length];
253	    }

    /// </summary>
    private static int LevenshteinDistance(string a, string b)
    {
        if (a.Length == 0) return b.Length;
        if (b.Length == 0) return a.Length;

        var prev = new int[b.Length + 1];
        var curr = new int[b.Length + 1];

        for (var j = 0; j <= b.Length; j++)
            prev[j] = j;

        for (var i = 1; i <= a.Length; i++)
        {
            curr[0] = i;
            for (var j = 1; j <= b.Length; j++)
            {
                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
                curr[j] = Math.Min(
                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
                    prev[j - 1] + cost);
            }

            (prev, curr) = (curr, prev);
        }

                curr[j] = Math.Min(
                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
                    prev[j - 1] + cost);

2	
3	**Source:** GitHub issue #11
4	**Scope:** Core / Host
5	**Complexity:** Medium
6	**Breaking changes:** None — existing exact-name behaviour is preserved; wildcards are additive
7	**Status:** Pending

9	---
10	
11	## Problem
12	
13	The `Plugins` configuration list requires every plugin to be named
14	individually. Downstream projects with many plugins find this tedious and
15	error-prone — adding a new plugin assembly to the plugin directory still
16	requires a config change. There is no way to say "load everything" or
17	"load everything except X".
18	
19	## Changes
20	
21	### 1. Add pattern expansion step in `PluginManager`
22	
23	Add a new static method to `PluginManager` (where `ResolveRequestedPlugins`
24	and `DeduplicateDirectories` already live). This method expands
25	wildcard/glob entries against the metadata-scanned plugin names:
26	
27	```csharp
28	internal static IReadOnlyList<string> ExpandPluginPatterns(
29	    IReadOnlyList<string> patterns,
30	    IReadOnlyList<PluginMetadata> allMetadata,
31	    ILogger? logger = null)
32	```
33	
34	**Pattern rules (evaluated left-to-right):**
35	
36	- A plain name (no `*` or `?` characters) is passed through as-is to
37	  `ResolveRequestedPlugins` for exact matching (preserving current
38	  behaviour including the "did you mean?" error).
39	- A glob pattern (contains `*` or `?`) is matched case-insensitively
40	  against all discovered plugin names using
41	  `FileSystemName.MatchesSimpleExpression`. Every match is added to
42	  the result set. An unmatched glob silently matches nothing (logged at
43	  Debug level).
44	- A negation pattern (prefixed with `!`) removes previously matched
45	  names. The remainder after `!` may be a plain name or a glob. For
46	  example, `["*", "!Slap"]` loads all plugins except `Slap`.
47	- Duplicate names are suppressed — each plugin appears at most once in
48	  the expanded result.
49	
50	The expanded list of concrete plugin names is then passed to
51	`ResolveRequestedPlugins` as today.
52	
53	### 2. Update `AddMarv` call site
54	
55	In `MarvServiceExtensions.AddMarv`, after the metadata scan and before
56	`PluginManager.ResolveRequestedPlugins`, call `ExpandPluginPatterns`:
57	
58	```csharp

60	    config.Plugins, allPluginMetadata, bootstrapLogger);

63	    expandedPlugins, allPluginMetadata, bootstrapLogger);
64	```
65	
66	### 3. Logging
67	
68	- Log at Information level when a glob pattern matches one or more
69	  plugins, listing the matched names.
70	- Log at Information level when a negation pattern excludes plugins.
71	- Log at Debug level when a glob matches nothing.
72	
73	### 4. Default behaviour unchanged
74	
75	An empty `Plugins` list (`[]`) still loads nothing. Users who want all
76	plugins must explicitly set `["*"]`. This avoids accidentally loading
77	untested or unwanted plugins.

92	## Design decisions
93	
94	- **`!` prefix for negation, not `-`:** The `!` prefix avoids ambiguity
95	  with YAML list syntax where `-` is a list item marker. This was
96	  confirmed by the issue author.
97	- **`FileSystemName.MatchesSimpleExpression`:** Available in
98	  `System.IO.Enumeration` without external dependencies. Supports `*`

102	  when unmatched, to catch typos), globs matching nothing is expected

104	  installed. Plain names retain the existing fatal-error-with-suggestions
105	  behaviour.
106	- **Left-to-right evaluation:** Simple and predictable. Matches how
107	  `.gitignore` and similar tools process patterns.

453	    /// </summary>
454	    private static int LevenshteinDistance(string a, string b)
455	    {
456	        if (a.Length == 0) return b.Length;
457	        if (b.Length == 0) return a.Length;
458	
459	        var prev = new int[b.Length + 1];
460	        var curr = new int[b.Length + 1];
461	
462	        for (var j = 0; j <= b.Length; j++)
463	            prev[j] = j;
464	
465	        for (var i = 1; i <= a.Length; i++)
466	        {
467	            curr[0] = i;
468	            for (var j = 1; j <= b.Length; j++)
469	            {
470	                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
471	                curr[j] = Math.Min(
472	                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
473	                    prev[j - 1] + cost);
474	            }

476	            (prev, curr) = (curr, prev);
477	        }

using Marv.Core.Plugin;
using Xunit;

namespace Marv.Core.Tests.Plugin;

/// <summary>
/// Tests for wildcard and negation pattern expansion in
/// <see cref="PluginManager.ExpandPluginPatterns"/>.
/// </summary>
public class PluginPatternExpansionTests
{
    private static PluginMetadata Meta(string name) =>
        new(name, $"/plugins/Marv.Plugins.{name}.dll", $"Marv.Plugins.{name}.dll");

    private static readonly IReadOnlyList<PluginMetadata> AllPlugins =
    [
        Meta("Auth"),
        Meta("Greet"),
        Meta("CannedResponses"),
        Meta("Moderation"),
        Meta("IdleRPG.Core"),
        Meta("IdleRPG.Combat"),
    ];

    [Fact]
    public void PlainNames_PassThroughUnchanged()
    {
        var result = PluginManager.ExpandPluginPatterns(["Auth", "Greet"], AllPlugins);

        Assert.Equal(2, result.Count);
        Assert.Equal("Auth", result[0]);
        Assert.Equal("Greet", result[1]);
    }

    [Fact]
    public void WildcardStar_ExpandsToAllPlugins()
    {
        var result = PluginManager.ExpandPluginPatterns(["*"], AllPlugins);

        Assert.Equal(AllPlugins.Count, result.Count);
        foreach (var meta in AllPlugins)
            Assert.Contains(meta.Name, result);
    }

    [Fact]
    public void GlobPattern_MatchesPrefix()
    {
        var result = PluginManager.ExpandPluginPatterns(["IdleRPG.*"], AllPlugins);

        Assert.Equal(2, result.Count);
        Assert.Contains("IdleRPG.Core", result);
        Assert.Contains("IdleRPG.Combat", result);
    }

    [Fact]
    public void QuestionMarkWildcard_MatchesSingleCharacter()
    {
        var result = PluginManager.ExpandPluginPatterns(["Gree?"], AllPlugins);

        Assert.Single(result);
        Assert.Equal("Greet", result[0]);
    }

    [Fact]
    public void Negation_ExcludesExactName()
    {
        var result = PluginManager.ExpandPluginPatterns(["*", "!Greet"], AllPlugins);

        Assert.Equal(AllPlugins.Count - 1, result.Count);
        Assert.DoesNotContain("Greet", result);
    }

    [Fact]
    public void Negation_ExcludesGlobPattern()
    {
        var result = PluginManager.ExpandPluginPatterns(["*", "!IdleRPG.*"], AllPlugins);

        Assert.Equal(AllPlugins.Count - 2, result.Count);
        Assert.DoesNotContain("IdleRPG.Core", result);
        Assert.DoesNotContain("IdleRPG.Combat", result);
    }

    [Fact]
    public void Negation_UnmatchedName_IsNoOp()
    {
        var result = PluginManager.ExpandPluginPatterns(["*", "!NonExistent"], AllPlugins);

        Assert.Equal(AllPlugins.Count, result.Count);
    }

    [Fact]
    public void UnmatchedGlob_MatchesNothing()
    {
        var result = PluginManager.ExpandPluginPatterns(["Foo.*"], AllPlugins);

        Assert.Empty(result);
    }

    [Fact]
    public void DuplicateNames_AreSuppressed()

        var result = PluginManager.ExpandPluginPatterns(["Auth", "Auth"], AllPlugins);

        Assert.Single(result);
        Assert.Equal("Auth", result[0]);
    }

    [Fact]
    public void GlobAndPlainName_DuplicateSuppressed()
    {

        Assert.Equal(AllPlugins.Count, result.Count);
        Assert.Equal("Auth", result[0]);
    }

    }

A single orchestrating method on PluginManager makes sense. The phases are all plugin-loading concerns — directory dedup, assembly resolvers, metadata scan, pattern expansion, name resolution, discover+register. MarvServiceExtensions shouldn't need to know the steps; it just needs the descriptors back.

Something like:

internal static IReadOnlyList<PluginDescriptor> ResolveAndRegister(
    IServiceCollection services,
    IConfiguration configuration,
    MarvConfiguration config,
    ILogger? logger = null)

AddMarv would reduce to: register core services, call PluginManager.ResolveAndRegister, store the result. The internal sub-methods (DeduplicateDirectories, ExpandPluginPatterns, ResolveRequestedPlugins, DiscoverAndRegister, RegisterAssemblyResolvers) stay as private helpers inside PluginManager — same decomposition, just no longer exposed.

The one piece that's arguably not PluginManager's job is RegisterAssemblyResolvers (it's about the AssemblyLoadContext, not plugin management), but it's only ever called as part of plugin loading and needs the plugin directories, so keeping it as a private detail of the pipeline seems cleaner than leaving it stranded in MarvServiceExtensions.
