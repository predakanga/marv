319	    /// Expands wildcard and negation patterns in the plugin list against
320	    /// discovered plugin metadata. Plain names pass through unchanged;
321	    /// glob patterns (<c>*</c>, <c>?</c>) match against plugin names;
322	    /// negation patterns (<c>!</c> prefix) remove previously matched names.
323	    /// Patterns are evaluated left-to-right.
324	    /// </summary>
325	    internal static IReadOnlyList<string> ExpandPluginPatterns(
326	        IReadOnlyList<string> patterns,
327	        IReadOnlyList<PluginMetadata> allMetadata,
328	        ILogger? logger = null)
329	    {
330	        var result = new List<string>();
331	        var resultSet = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
332	
333	        foreach (var pattern in patterns)
334	        {
335	            if (pattern.StartsWith('!'))
336	            {
337	                var inner = pattern[1..];
338	                if (IsGlobPattern(inner))
339	                {
340	                    var removed = result.RemoveAll(name =>
341	                        FileSystemName.MatchesSimpleExpression(inner, name, ignoreCase: true));
342	                    if (removed > 0)
343	                    {
344	                        resultSet.Clear();
345	                        foreach (var name in result)
346	                            resultSet.Add(name);
347	                        logger?.LogInformation(
348	                            "Negation pattern '{Pattern}' excluded {Count} plugin(s)",
349	                            pattern, removed);
350	                    }
351	                }
352	                else
353	                {
354	                    if (resultSet.Remove(inner))
355	                    {
356	                        result.RemoveAll(name =>
357	                            string.Equals(name, inner, StringComparison.OrdinalIgnoreCase));
358	                        logger?.LogInformation(
359	                            "Negation pattern '{Pattern}' excluded plugin '{Name}'",
360	                            pattern, inner);
361	                    }
362	                }
363	            }
364	            else if (IsGlobPattern(pattern))
365	            {
366	                var matched = new List<string>();
367	                foreach (var meta in allMetadata)
368	                {
369	                    if (FileSystemName.MatchesSimpleExpression(pattern, meta.Name, ignoreCase: true)
370	                        && resultSet.Add(meta.Name))
371	                    {
372	                        result.Add(meta.Name);
373	                        matched.Add(meta.Name);
374	                    }
375	                }
376	
377	                if (matched.Count > 0)
378	                    logger?.LogInformation(
379	                        "Pattern '{Pattern}' matched plugin(s): {Names}",
380	                        pattern, string.Join(", ", matched));
381	                else
382	                    logger?.LogDebug(
383	                        "Pattern '{Pattern}' matched no plugins", pattern);
384	            }
385	            else
386	            {
387	                if (resultSet.Add(pattern))
388	                    result.Add(pattern);
389	            }
390	        }
391	
392	        return result;
393	    }

421	    /// scan results. Matching order:
422	    /// 1. Exact plugin name match (case-insensitive).
423	    /// 2. Assembly filename convention match (e.g., "CannedResponses" matches
424	    ///    Marv.Plugins.CannedResponses.dll) — logs a warning.
425	    /// 3. No match — fatal error with suggestions.
426	    /// </summary>
427	    internal static IReadOnlyList<string> ResolveRequestedPlugins(
428	        IReadOnlyList<string> requestedPlugins,
429	        IReadOnlyList<PluginMetadata> allMetadata,
430	        ILogger? logger = null)
431	    {
432	        var resolvedPaths = new List<string>();
433	        var resolvedSet = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
434	
435	        foreach (var requested in requestedPlugins)
436	        {
437	            // 1. Exact plugin name match
438	            var exact = allMetadata.FirstOrDefault(
439	                m => string.Equals(m.Name, requested, StringComparison.OrdinalIgnoreCase));
440	
441	            if (exact is not null)
442	            {
443	                if (resolvedSet.Add(exact.AssemblyPath))

455	                string.Equals(

466	                    requested, conventionMatch.Name, conventionMatch.AssemblyPath,
467	                    conventionMatch.Name);
468	
469	                if (resolvedSet.Add(conventionMatch.AssemblyPath))
470	                    resolvedPaths.Add(conventionMatch.AssemblyPath);
471	                continue;
472	            }
473	
474	            // 3. No match — fatal error with suggestions
475	            var availableNames = allMetadata
476	                .Select(m => $"  - '{m.Name}' (from {m.AssemblyFileName})")
477	                .ToList();
478	
479	            var suggestion = FindClosestMatch(requested, allMetadata);
480	            var suggestionText = suggestion is not null
481	                ? $"\n\n  Did you mean '{suggestion}'?"
482	                : "";
483

485	                ? "\n\n  Available plugins in configured directories:\n" +
486	                  string.Join("\n", availableNames)
487	                : "\n\n  No plugins were found in the configured directories.";

491	                $"that name was found.{availableList}{suggestionText}");
492	        }

495	    }

497	    /// <summary>
498	    /// Finds the closest matching plugin name for a "did you mean?" suggestion.
499	    /// Uses case-insensitive substring matching and Levenshtein distance.
500	    /// </summary>
501	    private static string? FindClosestMatch(
502	        string requested, IReadOnlyList<PluginMetadata> allMetadata)
503	    {
504	        var requestedLower = requested.ToLowerInvariant();
505	        string? bestMatch = null;
506	        var bestDistance = int.MaxValue;
507	
508	        foreach (var meta in allMetadata)
509	        {
510	            // Check plugin name
511	            var distance = LevenshteinDistance(requestedLower, meta.Name.ToLowerInvariant());
512	            if (distance < bestDistance)
513	            {
514	                bestDistance = distance;
515	                bestMatch = meta.Name;
516	            }
517	
518	            // Check assembly-derived name
519	            var assemblyName = PluginMetadataScanner.DeriveNameFromAssemblyFile(meta.AssemblyFileName);
520	            distance = LevenshteinDistance(requestedLower, assemblyName.ToLowerInvariant());
521	            if (distance < bestDistance)
522	            {
523	                bestDistance = distance;
524	                bestMatch = meta.Name;
525	            }
526	        }
527	
528	        // Only suggest if the distance is reasonable (less than half the length)
529	        if (bestMatch is not null && bestDistance <= Math.Max(requested.Length, bestMatch.Length) / 2)
530	            return bestMatch;
531	
532	        return null;
533	    }

535	    /// <summary>
536	    /// Computes the Levenshtein edit distance between two strings.
537	    /// </summary>
538	    private static int LevenshteinDistance(string a, string b)
539	    {
540	        if (a.Length == 0) return b.Length;
541	        if (b.Length == 0) return a.Length;
542	
543	        var prev = new int[b.Length + 1];
544	        var curr = new int[b.Length + 1];
545	
546	        for (var j = 0; j <= b.Length; j++)
547	            prev[j] = j;
548	
549	        for (var i = 1; i <= a.Length; i++)
550	        {
551	            curr[0] = i;
552	            for (var j = 1; j <= b.Length; j++)
553	            {
554	                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
555	                curr[j] = Math.Min(
556	                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
557	                    prev[j - 1] + cost);
558	            }
559	
560	            (prev, curr) = (curr, prev);
561	        }
562	
563	        return prev[b.Length];
564	    }

        services.AddScoped<CapabilityManager>();

        services.AddHttpClient();
