**Purpose:** Organize handlers into separate classes for modularity

**Pattern:**

**`PluginDependencySorter`** (`/workspaces/marv/src/Marv.Core/Plugin/PluginDependencySorter.cs`):

Uses Kahn's topological sort algorithm to order plugins. Dependencies from:
1. `[ProvidesService]` / required constructor parameters
2. `[DependsOn]` explicit dependencies
3. Optional service dependencies (order-if-present)

- Validates required services have providers
- Detects and rejects cyclic dependencies
- Ensures providers load before consumers

using Marv.Core.Formatting;
using Marv.Core.Platform;
using Marv.Core.Plugin;
using Microsoft.Extensions.Logging;

namespace Marv.Plugins.Series;

/// <summary>
/// Looks up TV series information from a database and displays it in a formatted
/// IRC message. Requires <see cref="IPermissionService"/> and <see cref="ISeriesRepository"/>
/// to be provided by other plugins.
/// </summary>
public class SeriesPlugin : MarvPlugin
{
    private const int RequiredPermissionLevel = 100;

    private readonly IPermissionService? _permissions;
    private readonly ISeriesRepository? _series;
    private readonly ILogger<SeriesPlugin> _logger;

    /// <summary>
    /// Creates a new <see cref="SeriesPlugin"/>.
    /// </summary>
    public SeriesPlugin(
        IBot bot,
        IPluginActivator activator,
        ILogger<SeriesPlugin> logger,
        IPermissionService? permissions = null,
        ISeriesRepository? series = null)
        : base(bot, activator)
    {
        _logger = logger;
        _permissions = permissions;
        _series = series;
    }

    /// <summary>
    /// Handles the <c>!series</c> command. Looks up a series by name and displays
    /// its network, runtime, status, and community rating.
    /// </summary>
    [OnCommand("series")]
    [OnCommand("s")]
    private async Task HandleSeries(CommandContext ctx, CancellationToken ct)
    {
        if (string.IsNullOrWhiteSpace(ctx.ArgString))
        {
            await ctx.ReplyAsync("Usage: !series <name>", ct);
            return;
        }

        if (_permissions is null || _series is null)
        {
            _logger.LogWarning("Series plugin is missing required services (IPermissionService and/or ISeriesRepository)");
            await ctx.ReplyAsync("Series lookup is not available.", ct);
            return;
        }

        var level = await _permissions.GetPermissionLevelAsync(ctx.Sender, ct);
        _logger.LogDebug("!series cmd from: {Nick}, permission level: {Level}, series: {Series}",
            ctx.Sender.Nick, level, ctx.ArgString);

        if (level < RequiredPermissionLevel)
        {
            await ctx.ReplyAsync(
                "You don't have sufficient permissions to run this command or maybe you are not using your site nick?",
                ct);
            return;
        }

        var info = await _series.FindSeriesAsync(ctx.ArgString, ct);
        if (info is null)
        {
            await ctx.ReplyAsync($"Sorry, can't find series: {ctx.ArgString}", ct);
            return;
        }

        double? rating = await _series.GetAverageRatingAsync(info.Id, ct);

        var output = FormatSeriesOutput(info, rating);
        await ctx.ReplyAsync(output, ct);
    }

    private static string FormatSeriesOutput(SeriesInfo info, double? rating)
    {
        var status = FormatStatus(info);
        var ratingText = rating.HasValue ? $"{rating.Value:F2}" : "not available";

        // Reproduces the original IRC color formatting:
        // Cyan-on-Black brackets, Orange values, Green labels, Grey for URL
        return $"{IrcColor.Cyan.On(IrcColor.Black)}[ {IrcColor.Orange}{info.Name} " +
               $"{IrcColor.Cyan}] :: [ {IrcColor.Green}Network: {IrcColor.Orange}{info.Network ?? "N/A"} " +
               $"{IrcColor.Cyan}] :: [ {IrcColor.Green}Runtime: {IrcColor.Orange}{info.Runtime ?? "N/A"} " +
               $"{IrcColor.Cyan}] :: [ {IrcColor.Green}Status: {IrcColor.Orange}{status} " +
               $"{IrcColor.Cyan}] :: [ {IrcColor.Green}Rating: {IrcColor.Orange}{ratingText} " +
               $"{IrcColor.Cyan}]{IrcFormat.Reset}";
    }

    private static string FormatStatus(SeriesInfo info)
    {
        return info.Status switch
        {
            "Canceled/Ended" or "Ended" or "Canceled" => info.Status,
            "In Development" or "In Developmet" => "In Development",
            "Never Aired" => "Never Aired",
            "On Hiatus" => "On Hiatus",
            _ => info.AirDay is not null && info.AirTime is not null
                ? $"Airs: {info.AirDay} {info.AirTime}"
                : info.Status ?? "Unknown"
        };
    }
}
