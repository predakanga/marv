220	    /// </summary>
221	    private static int LevenshteinDistance(string a, string b)
222	    {
223	        if (a.Length == 0) return b.Length;
224	        if (b.Length == 0) return a.Length;
225	
226	        var prev = new int[b.Length + 1];
227	        var curr = new int[b.Length + 1];
228	
229	        for (var j = 0; j <= b.Length; j++)
230	            prev[j] = j;
231	
232	        for (var i = 1; i <= a.Length; i++)
233	        {
234	            curr[0] = i;
235	            for (var j = 1; j <= b.Length; j++)
236	            {
237	                var cost = a[i - 1] == b[j - 1] ? 0 : 1;
238	                curr[j] = Math.Min(
239	                    Math.Min(curr[j - 1] + 1, prev[j] + 1),
240	                    prev[j - 1] + cost);
241	            }
242	
243	            (prev, curr) = (curr, prev);
244	        }
245	
246	        return prev[b.Length];
247	    }
