430	    private async Task HandleCap(IrcMessage message, CancellationToken ct)
431	    {
432	        if (message.Parameters.Count < 3) return;
433	
434	        var subCommand = message.Parameters[1].ToUpperInvariant();
435	        var isMultiLine = message.Parameters.Count > 3 && message.Parameters[2] == "*";
436	        var capList = message.Parameters[isMultiLine ? 3 : 2];
437	
438	        switch (subCommand)
439	        {
440	            case "LS":
441	                foreach (var cap in ParseCapList(capList))
442	                {
443	                    _capabilityManager.SetAvailable(cap.Key, cap.Value);
444	                }
445	
446	                if (!isMultiLine)
447	                {
448	                    // All capabilities listed — request the ones we want
449	                    var toRequest = DesiredCapabilities
450	                        .Where(c => _capabilityManager.IsAvailable(c))
451	                        .ToList();
452	
453	                    if (toRequest.Count > 0)
454	                    {
455	                        _logger.LogInformation("Requesting capabilities: {Caps}",
456	                            string.Join(", ", toRequest));
457	                        await SendRawAsync(new IrcMessage("CAP", ["REQ", string.Join(' ', toRequest)]), ct);
458	                    }
459	                    else
460	                    {
461	                        await SendRawAsync(new IrcMessage("CAP", ["END"]), ct);
462	        
463	                    }
464	                }
465	                break;
466	
467	            case "ACK":
468	                foreach (var capName in capList.Split(' ', StringSplitOptions.RemoveEmptyEntries))
469	                {
470	                    _capabilityManager.SetNegotiated(capName);
471	                    _logger.LogDebug("Capability acknowledged: {Cap}", capName);
472	                }
473	
474	                // Start SASL if negotiated and credentials are configured
475	                if (_capabilityManager.IsNegotiated(Platform.Capabilities.Sasl)
476	                    && !string.IsNullOrEmpty(_config.SaslUser) && !string.IsNullOrEmpty(_config.SaslPassword)
477	                    && !_saslInProgress)
478	                {
479	                    _saslInProgress = true;
480	                    await SendRawAsync(new IrcMessage("AUTHENTICATE", ["PLAIN"]), ct);
481	                }
482	                else
483	                {
484	                    await SendRawAsync(new IrcMessage("CAP", ["END"]), ct);
485	    
486	                }
487	                break;
488	
489	            case "NAK":
490	                _logger.LogWarning("Capabilities rejected: {Caps}", capList);
491	                await SendRawAsync(new IrcMessage("CAP", ["END"]), ct);
492	
493	                break;
494	
495	            case "NEW":
496	                foreach (var cap in ParseCapList(capList))
497	                {
498	                    _capabilityManager.AddNewCapability(cap.Key, cap.Value);
499	                }
500	                var newCaps = ParseCapList(capList)
501	                    .Where(c => DesiredCapabilities.Contains(c.Key, StringComparer.OrdinalIgnoreCase))
502	                    .Select(c => c.Key)
503	                    .ToList();
504	                if (newCaps.Count > 0)
505	                {
506	                    await SendRawAsync(new IrcMessage("CAP", ["REQ", string.Join(' ', newCaps)]), ct);
507	                }
508	
509	                var capChangedNew = CreateCapChangedEvent(message);
510	                await FanOutEventAsync(capChangedNew, ct);
511	                break;
512	
513	            case "DEL":
514	                foreach (var capName in capList.Split(' ', StringSplitOptions.RemoveEmptyEntries))
515	                {
516	                    _capabilityManager.RemoveCapability(capName);
517	                    _logger.LogDebug("Capability removed: {Cap}", capName);
518	                }
519	
520	                var capChangedDel = CreateCapChangedEvent(message);
521	                await FanOutEventAsync(capChangedDel, ct);
522	                break;
523	        }
524	    }

648	    private async Task HandleCtcp(IrcMessage message, IrcUser sender, string target, string ctcpContent, CancellationToken ct)
649	    {
650	        var spaceIdx = ctcpContent.IndexOf(' ');
651	        var command = spaceIdx >= 0 ? ctcpContent[..spaceIdx].ToUpperInvariant() : ctcpContent.ToUpperInvariant();
652	        var args = spaceIdx >= 0 ? ctcpContent[(spaceIdx + 1)..] : null;
653	
654	        switch (command)
655	        {
656	            case "ACTION":
657	            {
658	                var isChannel = IsChannelName(target);
659	                var channel = isChannel ? GetChannel(target) : null;
660	                var evt = new ActionEvent
661	                {
662	                    Timestamp = GetTimestamp(message),
663	                    RawMessage = message,
664	                    MessageId = message.Tags.GetValueOrDefault("msgid"),
665	                    BatchId = message.Tags.GetValueOrDefault("batch"),
666	                    Channel = channel,
667	                    Sender = sender,
668	                    Text = args ?? ""
669	                };
670	                await FanOutEventAsync(evt, ct);
671	                break;
672	            }
673	            case "VERSION":
674	                await SendRawAsync(new IrcMessage("NOTICE", [sender.Nick,
675	                    "\x01VERSION Marv IRC Bot 0.1.0\x01"]), ct);
676	                break;
677	            case "PING":
678	                await SendRawAsync(new IrcMessage("NOTICE", [sender.Nick,
679	                    $"\x01PING {args ?? ""}\x01"]), ct);
680	                break;
681	            case "TIME":
682	                await SendRawAsync(new IrcMessage("NOTICE", [sender.Nick,
683	                    $"\x01TIME {DateTimeOffset.UtcNow:R}\x01"]), ct);
684	                break;
685	            default:
686	            {
687	                var evt = new CtcpEvent
688	                {
689	                    Timestamp = GetTimestamp(message),
690	                    RawMessage = message,
691	                    MessageId = message.Tags.GetValueOrDefault("msgid"),
692	                    BatchId = message.Tags.GetValueOrDefault("batch"),
693	                    Sender = sender,
694	                    Command = command,
695	                    Args = args,
696	                    IsDirect = !IsChannelName(target)
697	                };
698	                await FanOutEventAsync(evt, ct);
699	                break;
700	            }
701	        }
702	    }

    /// <summary>
    /// Registers a raw IRC client (not using IrcBot) with the given nick.
    /// Waits for the 001 welcome numeric before returning.
    /// </summary>
    private static async Task RegisterClientAsync(IrcConnection connection, string nick, CancellationToken ct)
    {
        await connection.Outbound.WriteAsync(new IrcMessage("NICK", [nick]), ct);
        await connection.Outbound.WriteAsync(new IrcMessage("USER", [nick, "0", "*", "Test User"]), ct);

        // Drain until we see 001 (RPL_WELCOME)
        await foreach (var msg in connection.Inbound.ReadAllAsync(ct))
        {
            if (msg.Command == "001")
                break;
        }
    }

    /// <summary>
    /// Sends a raw IRC line through the connection's outbound channel.
    /// </summary>
    private static async Task SendRawLineAsync(IrcConnection connection, string line, CancellationToken ct)
    {
        var msg = IrcParser.Parse(line);
        if (msg is not null)
            await connection.Outbound.WriteAsync(msg, ct);
    }

    /// <summary>
    /// Waits for a specific event type from the event channel reader.
    /// Returns null if the cancellation token fires first.
    /// </summary>
    private static async Task<T?> WaitForEventAsync<T>(ChannelReader<MarvEvent> reader, CancellationToken ct)
        where T : MarvEvent
    {
        await foreach (var evt in reader.ReadAllAsync(ct))
        {
            if (evt is T typed)
                return typed;
        }
        return null;
    }

    /// <summary>
    /// Reads from a connection's inbound channel until a message with a command
    /// containing the specified substring is found. Returns the serialized form.
    /// </summary>
    private static async Task<string> WaitForLineAsync(IrcConnection connection, string commandSubstring, CancellationToken ct)
    {
        await foreach (var msg in connection.Inbound.ReadAllAsync(ct))
        {
            var serialized = IrcSerializer.Serialize(msg);
            if (serialized.Contains(commandSubstring, StringComparison.OrdinalIgnoreCase))
                return serialized;
        }
        return "";
    }

    private static async Task<T?> WaitForEventAsync<T>(ChannelReader<MarvEvent> reader, CancellationToken ct)

    {
        await foreach (var evt in reader.ReadAllAsync(ct))

    /// <summary>

    private static async Task<string> WaitForLineAsync(IrcConnection connection, string commandSubstring, CancellationToken ct)
    {
